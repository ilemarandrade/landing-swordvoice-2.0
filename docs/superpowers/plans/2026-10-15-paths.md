# Paths (card, detalle público y logueado) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Card de path mejorada en "Nuestros Paths", página pública `/paths/[slug]` con selector entre paths y detalle con progreso en `/dashboard/paths/[slug]`, alimentados por la API real vía una capa HTTP global con axios + TanStack Query; además se elimina el envío de email.

**Architecture:** `src/common/api` pasa de `fetch` a una única instancia de axios (`apiClient`) detrás del mismo `request()` (que desempaqueta el sobre `{statusCode,message,data}` y valida con zod). La feature nueva `src/features/paths` tiene schemas, service, `queryOptions` reutilizables (servidor y cliente), hooks, `buildTrail` y los componentes. Las páginas públicas precargan en el servidor con `getQueryClient()` + `<HydrationBoundary>`; el dashboard consulta en el cliente.

**Tech Stack:** Next.js 15.5 App Router, React 19, TypeScript, Tailwind 3, axios, @tanstack/react-query 5, zod 4, Zustand.

**Spec:** `docs/superpowers/specs/2026-10-15-paths-design.md`

## Global Constraints

- Sigue `docs/architecture/conventions.md` (§3 dependencias, §4.1 componentes por carpeta, §4.8 rutas) y `.claude/rules/nextjs-app-router.md`.
- API: `NEXT_PUBLIC_API_URL`, local `http://localhost:4000/api/v1`; header `Accept-Language: es`. Sobre `{ statusCode, message, data }` / error `{ statusCode, message, errorCode, errors }`.
- Una sola instancia de axios: `apiClient`. `axios` solo se importa en `src/common/api/**`.
- Slug correcto `initiated` (no `iniciated`). Sin migración de sesión.
- Sin CTA "Elegir este path".
- Sin tests nuevos; `PathCard` test existente sigue verde; `httpClient.test.ts` se adapta a axios.
- Textos de UI en español. Solo Tailwind con tokens del repo (`primary`, `secondary`, `blurry`, `blue`, `grey`, `bg-ourPathsBackground`, `font-sedgwick`); grises del diseño como arbitrarios: `#E6E6EE`, `#C9C9D6`, `#9A9AAE`, superficie `#000E2799`.

- [ ] **Step 6: `request()` con axios** — reemplazar `src/common/api/httpClient.ts`:

```ts
import { isAxiosError, type AxiosResponse } from "axios";
import { z } from "zod";
import { ApiError } from "@/common/api/ApiError";
import { apiClient } from "@/common/api/axiosInstances";
import {
  GENERIC_ERROR_MESSAGE,
  INVALID_RESPONSE_MESSAGE,
  NETWORK_ERROR_MESSAGE,
} from "@/common/constants/messages";

type HttpMethod = "GET" | "POST" | "PUT" | "PATCH" | "DELETE";

type RequestOptions<TSchema extends z.ZodType> = {
  method?: HttpMethod;
  body?: unknown;
  params?: Record<string, string | number>;
  schema: TSchema;
};

export async function request<TSchema extends z.ZodType>(
  path: string,
  { method = "GET", body, params, schema }: RequestOptions<TSchema>,
): Promise<z.infer<TSchema>> {
  let response: AxiosResponse<unknown>;
  try {
    response = await apiClient.request<unknown>({
      url: path,
      method,
      data: body,
      params,
    });
  } catch (error) {
    if (isAxiosError(error) && error.response) {
      const { status, data } = error.response;
      throw new ApiError(
        status,
        readString(data, "message") ?? GENERIC_ERROR_MESSAGE,
        readString(data, "errorCode"),
      );
    }
    throw new ApiError(0, NETWORK_ERROR_MESSAGE);
  }

  const parsed = schema.safeParse(unwrapEnvelope(response.data));
  if (!parsed.success) {
    throw new ApiError(response.status, INVALID_RESPONSE_MESSAGE);
  }
  return parsed.data;
}

// La API responde { statusCode, message, data }: se valida solo `data`.
function unwrapEnvelope(payload: unknown): unknown {
  if (
    typeof payload === "object" &&
    payload !== null &&
    "statusCode" in payload &&
    "data" in payload
  ) {
    return payload.data;
  }
  return payload;
}

function readString(data: unknown, key: string): string | undefined {
  if (typeof data === "object" && data !== null && key in data) {
    const value: unknown = (data as Record<string, unknown>)[key];
    return typeof value === "string" ? value : undefined;
  }
  return undefined;
}
```

} from "@tanstack/react-query";
import { cache } from "react";

function makeQueryClient(): QueryClient {
  return new QueryClient({
    defaultOptions: {
      queries: { staleTime: 60_000 },
      mutations: { retry: 0 },
      dehydrate: {
        shouldDehydrateQuery: (query) =>
          defaultShouldDehydrateQuery(query) ||
          query.state.status === "pending",
      },
    },
  });
}

// Servidor: un cliente por petición (compartido entre generateMetadata y la página).
const getServerQueryClient = cache(makeQueryClient);
let browserQueryClient: QueryClient | undefined;

export function getQueryClient(): QueryClient {
  if (isServer) return getServerQueryClient();
  browserQueryClient ??= makeQueryClient();
  return browserQueryClient;
}

import { beforeEach, describe, expect, it, vi } from "vitest";
import { z } from "zod";
import { apiClient } from "@/common/api/axiosInstances";
import { request } from "@/common/api/httpClient";

vi.mock("@/common/api/axiosInstances", () => ({
  apiClient: { request: vi.fn() },
}));

const requestMock = vi.mocked(apiClient.request);
const okSchema = z.object({ ok: z.literal(true) });

function response(status: number, data: unknown): AxiosResponse {
  return { status, statusText: "", headers: {}, config: { headers: new AxiosHeaders() }, data };
}

function httpError(status: number, data: unknown) {
  return new AxiosError("fail", "ERR_BAD_RESPONSE", undefined, undefined, response(status, data));
}

describe("request", () => {
  beforeEach(() => requestMock.mockReset());

  it("sends method, body and params to the axios instance", async () => {
    requestMock.mockResolvedValue(response(200, { ok: true }));

    await request("/x", { method: "POST", body: { a: 1 }, params: { page: 2 }, schema: okSchema });
    expect(requestMock).toHaveBeenCalledWith({ url: "/x", method: "POST", data: { a: 1 }, params: { page: 2 } });
  });

  it("unwraps the API envelope and validates data", async () => {
    requestMock.mockResolvedValue(response(200, { statusCode: 200, message: "ok", data: { ok: true } }));

    await expect(request("/x", { schema: okSchema })).resolves.toEqual({ ok: true });
  });

  it("validates bodies without envelope", async () => {
    requestMock.mockResolvedValue(response(200, { ok: true }));

    await expect(request("/x", { schema: okSchema })).resolves.toEqual({ ok: true });
  });

  it("throws ApiError with message and errorCode on HTTP errors", async () => {
    requestMock.mockRejectedValue(
      httpError(404, { statusCode: 404, message: "El path no existe", errorCode: "PATH_NOT_FOUND", errors: null }),
    );

    await expect(request("/x", { schema: okSchema })).rejects.toMatchObject({
      name: "ApiError",
      status: 404,
      message: "El path no existe",
      errorCode: "PATH_NOT_FOUND",
    });
  });

  it("uses the generic message when the error body has no message", async () => {
    requestMock.mockRejectedValue(httpError(502, "boom"));

    await expect(request("/x", { schema: okSchema })).rejects.toMatchObject({ status: 502, message: "Ha ocurrido un error" });
  });

  it("throws ApiError when the data does not match the schema", async () => {
    requestMock.mockResolvedValue(response(200, { statusCode: 200, message: "ok", data: { ok: false } }));

    await expect(request("/x", { schema: okSchema })).rejects.toMatchObject({ status: 200, message: "Respuesta inválida del servidor" });
  });

  it("throws ApiError with status 0 on network failure", async () => {
    requestMock.mockRejectedValue(new AxiosError("Network Error", "ERR_NETWORK"));

    await expect(request("/x", { schema: okSchema })).rejects.toMatchObject({ status: 0, message: "No se pudo conectar con el servidor" });
  });

  title: z.string(),
  url: z.string().nullable(),
  description: z.string().nullable(),
  position: z.number(),
});

  id: z.string(),
  title: z.string(),
  description: z.string().nullable(),
  position: z.number(),
  isOptional: z.boolean(),
  certifications: z.array(topicCertificationSchema),
  subtopics: z.array(subtopicSchema),
});

  id: z.string(),
  name: z.string(),
  slug: z.string(),
  description: z.string().nullable(),
  currentVersion: z.number(),
});

export const pathDetailSchema = z.object({
  id: z.string(),
  name: z.string(),
  slug: z.string(),
  description: z.string().nullable(),
  version: z.number(),
  isCurrent: z.boolean(),
  requiredCertifications: z.array(requiredCertificationSchema),
  topics: z.array(topicSchema),

- [ ] **Step 4: `utils/buildTrail.ts`**

```ts
import type {
  PathDetail,
  PathProgress,
  RequiredCertification,
  Subtopic,
  Topic,
} from "@/features/paths/schemas/path.schema";

export type StepStatus = "done" | "current" | "pending" | null;

export type TrailTopicItem = {
  kind: "topic";
  topic: Topic;
  index: number;
  subtopics: Subtopic[];
  completedSubtopicIds: string[];
  completedCount: number;
  status: StepStatus;
};

export type TrailTrialItem = {
  kind: "trial";
  certification: RequiredCertification;
  index: number;
  status: StepStatus;
};

export type TrailItem = TrailTopicItem | TrailTrialItem;

export type Trail = {
  items: TrailItem[];
  topicCount: number;
  resourceCount: number;
  requiredCount: number;
  earnedCount: number;
  percent: number;
  isComplete: boolean;
};

const byPosition = (a: { position: number }, b: { position: number }) =>
  a.position - b.position;

export function buildTrail(path: PathDetail, progress?: PathProgress): Trail {
  const topics = [...path.topics].sort(byPosition);
  const earned = new Set(progress?.earnedCertificationIds ?? []);
  const completed = new Set(progress?.completedSubtopicIds ?? []);

  // Cada prueba requerida va después del último tema que la referencia.
  const lastTopicIndex = new Map<string, number>();
  topics.forEach((topic, index) =>
    topic.certifications.forEach((c) => lastTopicIndex.set(c.externalId, index)),
  );

  const items: TrailItem[] = [];
  let trialIndex = 0;
  let currentAssigned = false;
  const trialStatus = (c: RequiredCertification): StepStatus =>
    progress ? (earned.has(c.id) ? "done" : "pending") : null;
  const pushTrial = (certification: RequiredCertification) =>
    items.push({ kind: "trial", certification, index: trialIndex++, status: trialStatus(certification) });

  topics.forEach((topic, index) => {
    const subtopics = [...topic.subtopics].sort(byPosition);
    const completedSubtopicIds = subtopics.filter((s) => completed.has(s.id)).map((s) => s.id);
    const completedCount = completedSubtopicIds.length;
    let status: StepStatus = null;
    if (progress) {
      const certificationsDone = topic.certifications.every((c) => earned.has(c.id));
      if (completedCount === subtopics.length && certificationsDone) {
        status = "done";
      } else if (!currentAssigned && !topic.isOptional) {
        currentAssigned = true;
        status = "current";
      } else {
        status = "pending";
      }
    }
    items.push({ kind: "topic", topic, index, subtopics, completedSubtopicIds, completedCount, status });
    path.requiredCertifications
      .filter((c) => lastTopicIndex.get(c.externalId) === index)
      .forEach(pushTrial);
  });
  path.requiredCertifications
    .filter((c) => !lastTopicIndex.has(c.externalId))
    .forEach(pushTrial);

  const requiredCount = path.requiredCertifications.length;
  const earnedCount = path.requiredCertifications.filter((c) => earned.has(c.id)).length;
  const requiredSubtopics = topics.filter((t) => !t.isOptional).flatMap((t) => t.subtopics);
  const completedSubtopics = requiredSubtopics.filter((s) => completed.has(s.id)).length;
  const total = requiredSubtopics.length + requiredCount;

  return {
    items,
    topicCount: topics.length,
    resourceCount: topics.reduce((sum, t) => sum + t.subtopics.length, 0),
    requiredCount,
    earnedCount,
    percent: total ? Math.round(((completedSubtopics + earnedCount) / total) * 100) : 0,
    isComplete: progress !== undefined && requiredCount > 0 && earnedCount === requiredCount,
  };
}
```

import { usePathname } from "next/navigation";
import { dashboardNav } from "@/features/dashboard/constants/dashboardNav";

export function useActiveNavItem() {
  const pathname = usePathname();
  const activeItem =
    dashboardNav.find(
      (item) =>
        item.href === pathname ||
        (item.activePrefix !== undefined && pathname.startsWith(item.activePrefix)),
    ) ?? dashboardNav[0];
  return { activeItem };
}

```ts
import type { MetadataRoute } from "next";

import { routes } from "@/common/constants/routes";
import { SITE_URL } from "@/common/constants/site";
import { getPaths } from "@/features/paths";

type SitemapPage = {
  path: string;
  priority: number;
  changeFrequency: "monthly" | "yearly";
};

const pages: SitemapPage[] = [
  { path: routes.HOME, priority: 1, changeFrequency: "monthly" },
  { path: routes.PATHS, priority: 0.9, changeFrequency: "monthly" },
  { path: routes.REGISTER, priority: 0.8, changeFrequency: "monthly" },
  { path: routes.LOGIN, priority: 0.5, changeFrequency: "yearly" },
  { path: routes.TERMS, priority: 0.3, changeFrequency: "yearly" },
  { path: routes.PRIVACY, priority: 0.3, changeFrequency: "yearly" },
];

async function pathPages(): Promise<SitemapPage[]> {
  try {
    const { items } = await getPaths();
    return items.map((p) => ({ path: `${routes.PATHS}/${p.slug}`, priority: 0.9, changeFrequency: "monthly" }));
  } catch {
    // Si la API no responde, el sitemap sale sin los paths individuales.
    return [];
  }
}

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const all = [...pages, ...(await pathPages())];
  return all.map(({ path, priority, changeFrequency }) => ({
    url: `${SITE_URL}${path}`,
    lastModified: new Date(),
    changeFrequency,
    priority,
  }));
}
```
