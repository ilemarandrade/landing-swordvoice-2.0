**Goal:** Entrar o crear la cuenta con Google, GitHub o LinkedIn mediante el flujo authorization code (con `state` y PKCE S256 para Google y GitHub), con retorno en `/auth/callback`.

**Architecture:** La web arma la URL de autorización con los client IDs públicos y guarda el intento (`state`, proveedor, `code_verifier`) en `sessionStorage`. `/auth/callback` consume ese intento una sola vez, valida el `state` y canjea el `code` en `POST /auth/oauth/:provider`; con 200 guarda los tokens igual que el login. Todo con APIs nativas del navegador (`crypto`, `URLSearchParams`, `sessionStorage`).

**Tech Stack:** Next.js 15 App Router, React 19, TypeScript strict, axios (`apiClient`), TanStack Query 5, zod 4, Web Crypto, Vitest + Testing Library.

**Spec:** `docs/superpowers/specs/2026-10-19-oauth-design.md`

## Global Constraints

- Proveedores y parámetros exactos: Google `https://accounts.google.com/o/oauth2/v2/auth` `scope=openid email profile` PKCE; GitHub `https://github.com/login/oauth/authorize` `scope=read:user user:email` PKCE; LinkedIn `https://www.linkedin.com/oauth/v2/authorization` `scope=openid profile email` sin PKCE.
- `redirect_uri` = `window.location.origin + "/auth/callback"`, idéntica en la ida y en el `POST`.
- `codeVerifier` solo se envía para Google y GitHub.
- `NEXT_PUBLIC_GOOGLE_CLIENT_ID`, `NEXT_PUBLIC_GITHUB_CLIENT_ID`, `NEXT_PUBLIC_LINKEDIN_CLIENT_ID` opcionales; un valor vacío cuenta como no configurado. Sin client ID no hay botón.
- Intento en `sessionStorage` clave `sv-oauth`, consumido al leerlo, válido 10 minutos.
- `/auth/callback` fuera de `(guest)`, con `robots: noindex` y `referrer: "no-referrer"`; nunca toasts: todo el resultado se muestra en la pantalla.
- Sistema de diseño existente (`AuthStatus`, `SocialLoginButton`, `ButtonLink`, tokens de Tailwind). Textos de UI en español; imports con `@/`; sin `any`, `@ts-ignore`, `eslint-disable`.

```ts
function base64Url(bytes: Uint8Array): string {
  let binary = "";
  bytes.forEach((byte) => {
    binary += String.fromCharCode(byte);
  });
  return btoa(binary).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

// 32 bytes aleatorios → 43 caracteres del alfabeto que exige RFC 7636.
function randomBase64Url(): string {
  return base64Url(crypto.getRandomValues(new Uint8Array(32)));
}

export const createState = randomBase64Url;
export const createCodeVerifier = randomBase64Url;

export async function createCodeChallenge(verifier: string): Promise<string> {
  const digest = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(verifier));
  return base64Url(new Uint8Array(digest));
}
```

```ts
import { z } from "zod";
import { OAUTH_PROVIDER_IDS } from "@/features/auth/constants/oauthProviders";

const STORAGE_KEY = "sv-oauth";
const MAX_AGE_MS = 10 * 60 * 1000;

const attemptSchema = z.object({
  provider: z.enum(OAUTH_PROVIDER_IDS),
  state: z.string().min(1),
  codeVerifier: z.string().optional(),
  createdAt: z.number(),
});

export type OAuthAttempt = z.infer<typeof attemptSchema>;

export function saveOAuthAttempt(attempt: Omit<OAuthAttempt, "createdAt">): void {
  try {
    sessionStorage.setItem(STORAGE_KEY, JSON.stringify({ ...attempt, createdAt: Date.now() }));
  } catch {
    // Sin sessionStorage el callback dirá que el intento no es válido.
  }
}

// Se consume al leerlo: un `state` solo sirve para una vuelta del proveedor.
export function takeOAuthAttempt(): OAuthAttempt | null {
  try {
    const raw = sessionStorage.getItem(STORAGE_KEY);
    sessionStorage.removeItem(STORAGE_KEY);
    if (!raw) return null;
    const parsed = attemptSchema.safeParse(JSON.parse(raw));
    if (!parsed.success) return null;
    return Date.now() - parsed.data.createdAt > MAX_AGE_MS ? null : parsed.data;
  } catch {
    return null;
  }
}
```

```ts
// src/features/auth/utils/buildAuthorizeUrl.ts
import type { AvailableOAuthProvider } from "@/features/auth/constants/oauthProviders";

type AuthorizeOptions = { state: string; redirectUri: string; codeChallenge?: string };

export function buildAuthorizeUrl(
  provider: AvailableOAuthProvider,
  { state, redirectUri, codeChallenge }: AuthorizeOptions,
): string {
  const params = new URLSearchParams({
    client_id: provider.clientId,
    redirect_uri: redirectUri,
    response_type: "code",
    scope: provider.scope,
    state,
  });
  if (codeChallenge) {
    params.set("code_challenge", codeChallenge);
    params.set("code_challenge_method", "S256");
  }
  return `${provider.authorizeUrl}?${params.toString()}`;
}
```

```ts
import { useEffect, useState } from "react";
import type { AvailableOAuthProvider } from "@/features/auth/constants/oauthProviders";
import { buildAuthorizeUrl } from "@/features/auth/utils/buildAuthorizeUrl";
import { saveOAuthAttempt } from "@/features/auth/utils/oauthAttempt";
import { oauthRedirectUri } from "@/features/auth/utils/oauthRedirectUri";
import {
  createCodeChallenge,
  createCodeVerifier,
  createState,
} from "@/features/auth/utils/pkce";

async function goToProvider(provider: AvailableOAuthProvider) {
  const state = createState();
  const codeVerifier = provider.usesPkce ? createCodeVerifier() : undefined;
  const codeChallenge = codeVerifier ? await createCodeChallenge(codeVerifier) : undefined;
  saveOAuthAttempt({ provider: provider.id, state, codeVerifier });
  window.location.assign(
    buildAuthorizeUrl(provider, { state, redirectUri: oauthRedirectUri(), codeChallenge }),
  );
}

export function useStartOAuth() {
  const [isStarting, setIsStarting] = useState(false);

  // Al volver con "Atrás" el navegador restaura la página (bfcache) con los botones deshabilitados.
  useEffect(() => {
    const onPageShow = (event: PageTransitionEvent) => {
      if (event.persisted) setIsStarting(false);
    };
    window.addEventListener("pageshow", onPageShow);
    return () => window.removeEventListener("pageshow", onPageShow);
  }, []);

  return {
    start: (provider: AvailableOAuthProvider) => {
      setIsStarting(true);
      void goToProvider(provider);
    },
    isStarting,
  };
}
```

```ts
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { apiClient } from "@/common/api/apiClient";
import { authTokensSchema } from "@/common/api/auth.schema";
import { saveTokens } from "@/common/api/tokenStorage";
import type { OAuthProviderId } from "@/features/auth/constants/oauthProviders";

type OAuthSignInRequest = {
  provider: OAuthProviderId;
  code: string;
  redirectUri: string;
  codeVerifier?: string;
};

async function signInWithProvider({ provider, ...body }: OAuthSignInRequest) {
  const { data } = await apiClient.post(`/auth/oauth/${provider}`, body);
  return authTokensSchema.parse(data);
}

// Sin notifyError: la pantalla del callback muestra cualquier resultado.
export function useOAuthSignInMutation() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: signInWithProvider,
    onSuccess: (tokens) => {
      queryClient.clear();
      saveTokens(tokens);
    },
  });
}
```

- [ ] **Step 2 (developer): Crear `src/features/auth/utils/toOAuthErrorMessage.ts`**

```ts
import { ApiError } from "@/common/api/ApiError";
import { GENERIC_ERROR_MESSAGE } from "@/common/constants/messages";
import { oauthMessages } from "@/features/auth/constants/oauthMessages";
import { isApiErrorCode } from "@/features/auth/utils/isApiErrorCode";

export function toOAuthErrorMessage(error: unknown, providerLabel: string): string {
  if (isApiErrorCode(error, "INVALID_OAUTH_CODE")) return oauthMessages.INVALID_CODE;
  if (isApiErrorCode(error, "OAUTH_EMAIL_NOT_VERIFIED")) {
    return oauthMessages.emailNotVerified(providerLabel);
  }
  if (isApiErrorCode(error, "OAUTH_PROVIDER_UNAVAILABLE")) {
    return oauthMessages.unavailable(providerLabel);
  }
  return error instanceof ApiError ? error.message : GENERIC_ERROR_MESSAGE;
}
```

- [ ] **Step 3 (developer): Crear `src/features/auth/hooks/useOAuthCallback.ts`**

```ts
import { useRouter } from "next/navigation";
import { useEffect, useRef, useState } from "react";
import { routes } from "@/common/constants/routes";
import {
  oauthMessages,
  UNKNOWN_PROVIDER_LABEL,
} from "@/features/auth/constants/oauthMessages";
import { oauthProviders } from "@/features/auth/constants/oauthProviders";
import { useOAuthSignInMutation } from "@/features/auth/hooks/useOAuthSignInMutation";
import { takeOAuthAttempt } from "@/features/auth/utils/oauthAttempt";
import { oauthRedirectUri } from "@/features/auth/utils/oauthRedirectUri";
import { toOAuthErrorMessage } from "@/features/auth/utils/toOAuthErrorMessage";

export type OAuthCallbackParams = {
  code: string | null;
  state: string | null;
  error: string | null;
};

export function useOAuthCallback({ code, state, error }: OAuthCallbackParams) {
  const router = useRouter();
  const { mutateAsync } = useOAuthSignInMutation();
  // El code es de un solo uso y React (StrictMode) ejecuta los efectos dos veces en desarrollo.
  const hasRun = useRef(false);
  const [providerLabel, setProviderLabel] = useState(UNKNOWN_PROVIDER_LABEL);
  const [message, setMessage] = useState<string | null>(null);

  useEffect(() => {
    if (hasRun.current) return;
    hasRun.current = true;

    const attempt = takeOAuthAttempt();
    const provider = attempt ? oauthProviders[attempt.provider] : null;
    const label = provider?.label ?? UNKNOWN_PROVIDER_LABEL;
    setProviderLabel(label);

    if (error) {
      setMessage(
        error === "access_denied"
          ? oauthMessages.cancelled(label)
          : oauthMessages.providerError(label),
      );
      return;
    }
    if (!attempt || !provider || !code || state !== attempt.state) {
      setMessage(oauthMessages.INVALID_ATTEMPT);
      return;
    }

    // mutateAsync (y no callbacks de mutate) para que el resultado llegue aunque StrictMode remonte.
    mutateAsync({
      provider: attempt.provider,
      code,
      redirectUri: oauthRedirectUri(),
      ...(provider.usesPkce && attempt.codeVerifier
        ? { codeVerifier: attempt.codeVerifier }
        : {}),
    }).then(
      () => router.replace(routes.DASHBOARD),
      (mutationError: unknown) => setMessage(toOAuthErrorMessage(mutationError, label)),
    );
  }, [code, state, error, mutateAsync, router]);

  return { providerLabel, message };
}
```
